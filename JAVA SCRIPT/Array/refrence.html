JavaScript Array References

1. Array Reference

Two variables can refer to the same array.

let arr1 = [10, 20, 30];
let arr2 = arr1;

arr2.push(40);

console.log(arr1); // [10, 20, 30, 40]
console.log(arr2); // [10, 20, 30, 40]

2. Changing Array Elements

let a = [1, 2, 3];
let b = a;

b[0] = 100;

console.log(a); // [100, 2, 3]
console.log(b); // [100, 2, 3]

3. Copy Array Using Spread Operator

let a = [1, 2, 3];
let b = [...a];

b[0] = 100;

console.log(a); // [1, 2, 3]
console.log(b); // [100, 2, 3]

4. Copy Array Using slice()

let a = [1, 2, 3];
let b = a.slice();

b.push(4);

console.log(a); // [1, 2, 3]
console.log(b); // [1, 2, 3, 4]

5. Comparing Arrays

let a = [1, 2, 3];
let b = [1, 2, 3];
let c = a;

console.log(a === b); // false
console.log(a === c); // true

6. Shallow Copy with Nested Arrays

let original = [[1, 2], [3, 4]];
let copy = [...original];

copy[0].push(99);

console.log(original); // [[1, 2, 99], [3, 4]]
console.log(copy);     // [[1, 2, 99], [3, 4]]
